cmake_minimum_required(VERSION 3.22.1)
project(stringer CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(oboe REQUIRED CONFIG)
add_library(stringer SHARED native.cpp)
target_include_directories(stringer PRIVATE engine)
# the audio code must be optimised even in a debug build of the app
target_compile_options(stringer PRIVATE -O3 -Wall -Wno-misleading-indentation)
target_link_libraries(stringer oboe::oboe log)
